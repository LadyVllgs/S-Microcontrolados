// aqui se definen los LEDS
#define LED1 5
#define LED2 18
#define LED3 19

// se define el pin del potenciómetro
#define POT 34   

void setup() {
  pinMode(LED1, OUTPUT);
  pinMode(LED2, OUTPUT);
  pinMode(LED3, OUTPUT);
  Serial.begin(115200);
}

void loop() {
  int valor = analogRead(POT); 
  int lectura = map(valor, 0, 4095, 0, 1023); // se escala de 4095 (12 bits) a 1023 (10 bits)
  Serial.print("ADC Lectura: ");
  Serial.print(valor);
  Serial.print(" -> Escalado: ");
  Serial.println(lectura);

  
  digitalWrite(LED1, LOW);
  digitalWrite(LED2, LOW);
  digitalWrite(LED3, LOW);

  // condiciones
  if (lectura > 767) {
    digitalWrite(LED1, HIGH);
    digitalWrite(LED2, HIGH);
    digitalWrite(LED3, HIGH);
  } else if (lectura >= 512) {
    digitalWrite(LED1, HIGH);
    digitalWrite(LED2, HIGH);
  } else if (lectura >= 256) {
    digitalWrite(LED1, HIGH);
  }
  
  delay(100); 
}
